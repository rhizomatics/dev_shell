97 async def run( 

98 self, 

99 source: str, 

100 timeout: float | None = None, 

101 *, 

102 color: bool = False, 

103 width: int = 88, 

104 auto_await: bool = True, 

105 ) -> ExecResult: 

106 """Execute source in this session, returning captured output and the last value.""" 

107 async with self._lock: 

108 self.last_used = time.time() 

109 self.executions += 1 

110 out = io.StringIO() 

111 # hass/obj/open are re-seeded every run, not just at session creation, so a 

112 # snippet that does `obj = obj["/mqtt"]["..."]` only shadows them for its own 

113 # run - the next command always starts from the real bindings again. 

114 self.globals_.update(self.protected) 

115 self.globals_["print"] = _capturing_print(out) 

116 self.globals_["help"] = _capturing_help(out, color=color, width=width) 

117 self.globals_["_maybe_await"] = _maybe_await 

118 self.globals_["unawait"] = unawait 

119 result = ExecResult() 

120 start = time.perf_counter() 

121 try: 

122 coro = self._execute(source, auto_await=auto_await) 

123 value = await (asyncio.wait_for(coro, timeout) if timeout else coro) 

124 if value is not None: 

125 self.globals_["_"] = value 

126 arrow_bytes = _maybe_arrow(value) 

127 if arrow_bytes is not None: 

128 # This value downloaded its own data (sql()'s 

129 # SqlResult today, duck-typed rather than 

130 # isinstance-checked - see _maybe_arrow()) - the 

131 # client renders *that*, not a server-side repr. 

132 result.arrow = base64.b64encode(arrow_bytes).decode("ascii") 

133 result.arrow_truncated = bool( 

134 getattr(value, "truncated", False) 

135 ) 

136 result.value = repr(value) 

137 else: 

138 # A value that already knows how to render itself as 

139 # a rich renderable is shown as-is instead of being 

140 # wrapped in Pretty's generic repr-style rendering, 

141 # which would just dump its attributes instead of 

142 # rendering it properly. 

143 renderable = ( 

144 value 

145 if hasattr(value, "__rich_console__") 

146 else Pretty(value) 

147 ) 

148 result.value = _render(renderable, color=color, width=width) 

149 except asyncio.CancelledError: 

150 # Cancellation of the caller must propagate, not be reported as a result. 

151 raise 

152 except BaseException as err: # noqa: BLE001 - report everything, incl. SystemExit 

153 result.error = _format_error(err, color=color, width=width) 

154 finally: 

155 result.duration = time.perf_counter() - start 

156 result.stdout = out.getvalue() 

157 for attr in ("stdout", "value"): 

158 text = getattr(result, attr) 

159 if text is not None and len(text) > MAX_OUTPUT_CHARS: 159 ↛ 160line 159 didn't jump to line 160 because the condition on line 159 was never true

160 setattr(result, attr, text[:MAX_OUTPUT_CHARS]) 

161 result.truncated = True 

162 return result

164 async def _execute(self, source: str, *, auto_await: bool = True) -> Any: 

165 # Not "<ha_repl-N>": rich.traceback refuses to show source for any 

166 # filename starting with "<" (treats it like "<stdin>"), no matter what 

167 # linecache holds. An absolute-looking path sidesteps that - rich joins a 

168 # relative one onto the cwd before the linecache lookup, which would miss. 

169 filename = f"/ha_repl/cell_{next(_cell_counter)}" 

170 # Register the source so tracebacks can show the offending lines. 

171 linecache.cache[filename] = ( 

172 len(source), 

173 None, 

174 # Always newline-terminated: rich's traceback rendering fails 

175 # ("substring not found") on a source with no newline in it at 

176 # all, i.e. any one-line command. 

177 (source if source.endswith("\n") else source + "\n").splitlines( 

178 keepends=True 

179 ), 

180 filename, 

181 ) 

182 tree = ast.parse(source, filename, "exec") 

183 if auto_await: 

184 tree = _AutoAwait().visit(tree) 

185 ast.fix_missing_locations(tree) 

186 

187 # Like the interactive interpreter, echo the value of a trailing expression. 

188 last_expr = None 

189 last_stmt = tree.body[-1] if tree.body else None 

190 if isinstance(last_stmt, ast.Expr): 

191 tree.body.pop() 

192 last_expr = ast.Expression(last_stmt.value) 

193 

194 flags = ast.PyCF_ALLOW_TOP_LEVEL_AWAIT 

195 # dont_inherit=True: compile() otherwise inherits this module's own 

196 # `from __future__ import annotations`, which would make every 

197 # annotation in the user's code a plain string - breaking the 

198 # FORWARDREF-based signature rendering in _format_signature below. 

199 if tree.body: 

200 await _run_code( 

201 compile(tree, filename, "exec", flags=flags, dont_inherit=True), 

202 self.globals_, 

203 ) 

204 if last_expr is None: 

205 return None 

206 return await _run_code( 

207 compile(last_expr, filename, "eval", flags=flags, dont_inherit=True), 

208 self.globals_, 

209 )

229def _maybe_arrow(value: Any) -> bytes | None: 

230 """`value.arrow()`'s bytes, if it has a no-arg method by that name - 

231 duck-typed, not isinstance-checked, so this module stays free of 

232 Home Assistant/sql.py imports (see this module's own docstring). Only 

233 sql.py's SqlResult exposes this today, but nothing here hard-codes 

234 that: anything that downloads its own data and wants the client to 

235 render it from that, rather than from a server-side repr, just needs

239 if not callable(method): 

240 return None 

241 try: 

242 data = method() 

243 except Exception: # noqa: BLE001 - fall back to a normal repr, not a crash 

244 return None 

245 return data if isinstance(data, bytes) else None

276def unawait(value: Any) -> Any: 

277 """Identity function and auto-await escape hatch: `unawait(f())` returns 

278 f()'s bare, un-awaited result (a coroutine, if f is async) - for when 

279 that's genuinely wanted, e.g. batching into `asyncio.gather(*[unawait(f())

281 which skips its whole argument rather than calling this at runtime; this

288class _AutoAwait(ast.NodeTransformer): 

289 """Rewrites every call not already explicitly awaited to go through 

290 _maybe_await() first, so `hass.async_foo()` works whether or not the 

291 user remembered `await` - including nested inside attribute access 

292 (`hass.async_foo().attr`), a comprehension, or an argument list. Leaves 

293 nested (synchronous) function/lambda bodies alone: `await` there is a 

294 SyntaxError, and those calls run later, not as part of this statement 

295 anyway. `unawait(expr)` is the escape hatch - its argument is left 

296 completely untouched for when the bare coroutine is wanted. Opt out 

297 entirely with `auto_await=False` on the exec call (strict mode: 

298 forgetting await behaves exactly as in component code). 

299 """

307 def visit_Await(self, node: ast.Await) -> ast.Await: 

308 # Already explicit: don't double-await this call, but still rewrite 

309 # anything nested inside its own arguments.

311 node.value.func = self.visit(node.value.func) 

312 node.value.args = [self.visit(a) for a in node.value.args] 

313 node.value.keywords = [self.visit(k) for k in node.value.keywords] 

314 else: 

315 node.value = self.visit(node.value) 

316 return node

319 if (

333async def _run_code(code: Any, globals_: dict[str, Any]) -> Any: 

334 # No separate "bare coroutine" backstop here: when auto_await is on, the 

335 # _AutoAwait rewrite already resolves every call site, and NOT doing so 

336 # unconditionally is exactly what `unawait(...)` asks for - a backstop 

337 # here would silently defeat it for a trailing `unawait(f())`. 

338 result = eval(code, globals_) # nosec B307 - the whole point of a dev shell 

339 if code.co_flags & inspect.CO_COROUTINE: 

340 result = await result 

341 return result

402_SIGNATURE_LEXER = PythonLexer() 

403# rich.syntax.Syntax itself renders as a block (its own padding/background 

404# handling, one truecolor SGR sequence per token even with 

405# background_color="default") - fine stacked one-at-a-time in a real 

406# terminal, but composed many-to-a-page inside a Group it came out jumbled 

407# for a class with lots of methods. Lexing with pygments directly and 

408# building a plain Text keeps this a single clean inline run per line, no 

409# block rendering involved, and ansi_dark's named 16-colour styles 

410# (`bright_cyan` etc.) over Syntax's default truecolor theme travel through 

411# a websocket/terminal pair more predictably.

415def _signature_line(line: str, *, width: int) -> Text: 

416 """A class/method signature as a one-line syntax-highlighted Text 

417 (pygments' Python lexer tokenizes it fine even though, as a bare 

418 "name(args) -> ret" fragment, it isn't valid standalone Python) - so 

419 type annotations, defaults and operators get real highlighting instead 

420 of being one flat-coloured string. 

421 

422 Home Assistant's own methods lean hard on generics (Callable[[Unpack[ 

423 _Ts]], ...]), so a real signature routinely runs past any reasonable 

424 width - wrapped here with a hanging indent *before* lexing (so the 

425 inserted newlines/spaces are just more whitespace tokens to pygments) 

426 rather than left to Rich's own word-wrap, which breaks at the console 

427 width with no indent at all and reads as a jumble of unrelated lines 

428 once there are dozens of methods back to back. 

429 """ 

430 wrapped = "\n".join( 

431 textwrap.wrap( 

432 line, 

433 width=max(width, 20), 

434 subsequent_indent=" ", 

435 break_long_words=False, 

436 break_on_hyphens=False, 

437 ) 

438 ) 

439 tokens = list(lex(wrapped, _SIGNATURE_LEXER)) 

440 while tokens and not tokens[-1][1].strip(): 

441 tokens.pop() # pygments always appends a trailing "\n" token 

442 text = Text(no_wrap=True) 

443 for token_type, value in tokens: 

444 text.append(value, style=_SIGNATURE_THEME.get_style_for_token(token_type)) 

445 return text

448def _class_summary(thing: Any, *, color: bool, width: int) -> str: 

449 """A class's full pydoc page repeats its docstring once per method, which 

450 balloons for a class like HomeAssistant with hundreds of methods. A method's 

451 own docstring is one `help(hass.the_method)` away, so here we only show the 

452 class docstring, the constructor signature, and the methods' names and 

453 signatures (no live attribute values either - that's what `hass.foo` is for). 

454 """ 

455 cls = thing if inspect.isclass(thing) else type(thing) 

456 header = ( 

457 f"Help on class {cls.__qualname__} in module {cls.__module__}:" 

458 if inspect.isclass(thing) 

459 else f"Help on {cls.__qualname__} object in module {cls.__module__}:" 

460 ) 

461 parts: list[Any] = [Text(header, style="bold"), Text("")] 

462 doc = inspect.getdoc(cls) 

463 if doc: 

464 parts += [Text(doc), Text("")] 

465 # A constructor "returning Self" is implied, not useful to state. 

466 ctor_sig = f"{cls.__qualname__}{_format_signature(cls, drop_return=(typing.Self,))}" 

467 parts.append(_signature_line(ctor_sig, width=width)) 

468 method_names = sorted( 

469 name 

470 for name in dir(cls) 

471 if not name.startswith("_") and _is_plain_method(getattr(cls, name, None)) 

472 ) 

473 if method_names: 

474 parts += [Text(""), Text("Methods:", style="bold underline")] 

475 for name in method_names: 

476 sig = _format_signature(getattr(cls, name), drop_self=True) 

477 parts.append(_signature_line(f" {name}{sig}", width=width)) 

478 property_names = sorted( 

479 name 

480 for name in dir(cls) 

481 if not name.startswith("_") and isinstance(getattr(cls, name, None), property) 

482 ) 

483 if property_names: 

484 parts += [Text(""), Text("Properties:", style="bold underline")] 

485 for name in property_names: 

486 doc = inspect.getdoc(getattr(cls, name)) 

487 summary = doc.strip().splitlines()[0] if doc else "" 

488 line = f" {name}" + (f" - {summary}" if summary else "") 

489 parts.append( 

490 Text( 

491 textwrap.fill( 

492 line, width=max(width, 20), subsequent_indent=" " 

493 ) 

494 ) 

495 ) 

496 return _render(Group(*parts), color=color, width=width)

500 # inspect.isroutine() also matches any non-data descriptor (defines __get__ but

508# Matches a dotted path ending in an identifier, e.g. "homeassistant.core.State" 

509# or the "collections.abc.Coroutine" inside a bigger type expression - used to 

510# shorten type annotations down to just the class name. 

511_DOTTED_NAME = re.compile(r"\b(?:[A-Za-z_]\w*\.)+([A-Za-z_]\w*)\b") 

512 

513# Python 3.14+ (PEP 649): __annotations__ access eagerly evaluates every annotation 

514# by default, which raises for names only imported under TYPE_CHECKING - common in 

515# HA's own codebase (e.g. Entity methods taking an "EntityPlatform"). FORWARDREF 

516# format resolves what it can and leaves the rest as an inert placeholder instead 

517# of raising. Not available before 3.14, hence the getattr dance. 

518_FORWARDREF_FORMAT = getattr(getattr(inspect, "Format", None), "FORWARDREF", None)

521def _format_signature( 

522 target: Any, *, drop_self: bool = False, drop_return: tuple = () 

523) -> str: 

524 """Render target's signature, trimmed down for a quick overview: 

525 self dropped (when `drop_self`), a void or otherwise uninformative return 

526 annotation dropped (`drop_return`), type annotations shortened to their bare 

527 class name, and no padding around `=` for default values. 

528 """ 

529 sig = None 

530 if _FORWARDREF_FORMAT is not None: 530 ↛ 535line 530 didn't jump to line 535 because the condition on line 530 was always true

531 try: 

532 sig = inspect.signature(target, annotation_format=_FORWARDREF_FORMAT) 

533 except Exception: # noqa: BLE001 - fall through to the attempts below 

534 sig = None 

535 if sig is None: 535 ↛ 536line 535 didn't jump to line 536 because the condition on line 535 was never true

536 try: 

537 sig = inspect.signature(target, eval_str=True) 

538 except Exception: # noqa: BLE001 - unresolvable forward refs can raise almost 

539 # anything (NameError, AttributeError, ...); a signature with raw/partial 

540 # annotations is still better than losing the method entirely. 

541 try: 

542 sig = inspect.signature(target) 

543 except TypeError, ValueError: 

544 return "(...)" 

545 params = list(sig.parameters.values()) 

546 if drop_self and params and params[0].name == "self": 

547 sig = sig.replace(parameters=params[1:]) 

548 ret = sig.return_annotation 

549 if ret is not sig.empty and ( 

550 ret is None or ret is type(None) or any(ret is d for d in drop_return) 

551 ): 

552 sig = sig.replace(return_annotation=sig.empty) 

553 text = _DOTTED_NAME.sub(r"\1", str(sig)) 

554 return re.sub(r"\s*=\s*", "=", text)

557def _format_error(err: BaseException, *, color: bool, width: int) -> dict[str, str]: 

558 tb = err.__traceback__ 

559 # Drop the frames belonging to this module so the traceback starts at user code. 

560 while tb is not None and tb.tb_frame.f_code.co_filename == __file__: 

561 tb = tb.tb_next 

562 if isinstance(err, SyntaxError): 

563 # No frames worth showing for this one, just the offending line and caret - 

564 # a plain rendering already does that job, so it skips the Rich treatment. 

565 text = "".join(traceback.format_exception_only(type(err), err)) 

566 else: 

567 text = _render( 

568 Traceback.from_exception(type(err), err, tb, width=width), 

569 color=color, 

570 width=width, 

571 ) 

572 return { 

573 "type": type(err).__name__, 

574 "message": str(err), 

575 "traceback": text, 

576 }

579@dataclass(frozen=True) 

580class PerSession: 

581 """Wraps a zero-arg factory for a binding that needs its own instance in 

582 every session - e.g. sql's mutable `.limit`, which would otherwise leak 

583 between sessions since `hass`/`obj` and friends are deliberately one 

584 shared instance for all of them. SessionManager.get() calls the factory 

585 once, the first time each session is created. 

586 """ 

587 

588 factory: Callable[[], Any] 

589 

590 

591class SessionManager: 

592 """Holds sessions by name; a session lives until reset or process restart.""" 

593 

594 def __init__(self, bindings: dict[str, Any]) -> None: 

595 self._bindings = bindings 

596 self._sessions: dict[str, Session] = {} 

597 

598 def get(self, name: str) -> Session: 

599 if (session := self._sessions.get(name)) is None: 

600 resolved = { 

601 k: (v.factory() if isinstance(v, PerSession) else v) 

602 for k, v in self._bindings.items() 

603 } 

604 globals_ = { 

605 "__name__": "__ha_repl__", 

606 "__builtins__": builtins, 

607 **resolved, 

608 } 

609 session = self._sessions[name] = Session(name, globals_, resolved) 

610 return session 

611 

612 def has_binding(self, name: str) -> bool: 

613 """Whether this name is bound into sessions at all (e.g. "sql" is 

614 left out when the integration's expose_sql option is off) - for the 

615 websocket commands that serve a binding's data directly rather than 

616 through exec, to honour the same switch.""" 

617 return name in self._bindings 

618 

619 def reset(self, name: str) -> bool: 

620 return self._sessions.pop(name, None) is not None 

621 

622 def describe(self) -> list[dict[str, Any]]: 

623 return [ 

624 { 

625 "name": s.name, 

626 "created": s.created, 

627 "last_used": s.last_used, 

628 "executions": s.executions, 

629 "variables": sorted( 

630 k 

631 for k in s.globals_ 

632 if not k.startswith("__") 

633 and k not in ("print", "help", "_maybe_await", "unawait") 

634 ), 

635 } 

636 for s in self._sessions.values() 

637 ]
